cmake_minimum_required(VERSION 3.24)
project(BloxdNativeRenderer VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32)
  message(FATAL_ERROR "The initial native prototype targets Windows.")
endif()
include(FetchContent)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glfw
  URL "${CMAKE_CURRENT_SOURCE_DIR}/third_party/glfw-3.4.zip"
  URL_HASH SHA256=a133ddc3d3c66143eba9035621db8e0bcf34dba1ee9514a9e23e96afd39fd57a
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(glfw)
FetchContent_Declare(glm
  URL "${CMAKE_CURRENT_SOURCE_DIR}/third_party/glm-1.0.1.zip"
  URL_HASH SHA256=09c5716296787e1f7fcb87b1cbdbf26814ec1288ed6259ccd30d5d9795809fa5
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(glm)
add_executable(BloxdNative src/main.cpp)
target_compile_features(BloxdNative PRIVATE cxx_std_20)
target_compile_definitions(BloxdNative PRIVATE GLFW_INCLUDE_NONE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(BloxdNative PRIVATE third_party/include)
target_link_libraries(BloxdNative PRIVATE glfw glm::glm windowscodecs ole32)
if(MSVC)
  target_compile_options(BloxdNative PRIVATE /W4 /utf-8 /permissive-)
endif()
enable_testing()
add_executable(MemoryBudgetRunner src/memory_budget_runner.cpp)
add_executable(MemoryBudgetTests EXCLUDE_FROM_ALL src/memory_budget_tests.cpp)
foreach(memory_target MemoryBudgetRunner MemoryBudgetTests)
  target_compile_features(${memory_target} PRIVATE cxx_std_20)
  target_compile_definitions(${memory_target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(${memory_target} PRIVATE psapi)
  if(MSVC)
    target_compile_options(${memory_target} PRIVATE /W4 /utf-8 /permissive-)
  endif()
endforeach()
add_test(NAME native_gpu_smoke COMMAND BloxdNative --self-test --output "${CMAKE_CURRENT_SOURCE_DIR}/outputs")
set_tests_properties(native_gpu_smoke PROPERTIES TIMEOUT 60)

# Explicitly built CPU-only contract test; never creates a graphics context.
add_executable(BridgeCpuTests EXCLUDE_FROM_ALL src/bridge_cpu_tests.cpp)
target_compile_features(BridgeCpuTests PRIVATE cxx_std_20)
target_compile_definitions(BridgeCpuTests PRIVATE GLFW_INCLUDE_NONE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(BridgeCpuTests PRIVATE third_party/include)
target_link_libraries(BridgeCpuTests PRIVATE glfw glm::glm)
if(MSVC)
  target_compile_options(BridgeCpuTests PRIVATE /W4 /utf-8 /permissive-)
endif()
